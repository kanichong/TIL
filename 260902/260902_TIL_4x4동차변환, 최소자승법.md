   측정한 값의 차이 = 잔차
   r = b - Ax
   A^t x - b
   
   A^t Ax = A^t b
   ||Ax -b||^2
   r = Ax -b
   ||r||^2 = r^t r
   ||Ax -b|| = (Ax - b)^t (Ax)
   
   x^t A^t Ax - 2x^t A^t b + b^t b
   x에 대해 미분
   0 = 2A^t Ax -2A^t b
   2A^t Ax = 2A^t b
   A^t Ax = A^t b
   
   r = b - Ax
   A^t r = 0
